  const pageKey = 'eng_lesson_' + window.location.pathname;

  function saveField(el){
    const elId = el.id || el.name;
    if (!elId) return;
    let value = el.type === 'checkbox' || el.type === 'radio' ? el.checked : el.value;
    localStorage.setItem(pageKey + '_' + elId, JSON.stringify(value));
  }

  function restoreFields(){
    document.querySelectorAll('input, textarea, select').forEach(el => {
      const elId = el.id || el.name;
      if (!elId) return;
      const saved = localStorage.getItem(pageKey + '_' + elId);
      if (saved !== null) {
        const value = JSON.parse(saved);
        el.type === 'checkbox' || el.type === 'radio' ? el.checked = value : el.value = value;
      }
    });
    updateProgress();
  }

  function updateProgress() {
    let filled = 0, total = 0;
    const radios = document.querySelectorAll('input[type="radio"]');
    const radioGroups = new Set();
    radios.forEach(r => {
      radioGroups.add(r.name);
      if (r.checked && r.classList.contains('corr')) filled++;
    });
    total += radioGroups.size;
    const trackers = document.querySelectorAll('.progress-tracker');
    total += trackers.length;
    trackers.forEach(tracker => { if (tracker.dataset.status === 'correct') filled++; });
    const percent = total > 0 ? (filled / total) * 100 : 0;
    const progressBar = document.getElementById('progressBar');
    const progressText = document.getElementById('progressText');
    if (progressBar) progressBar.style.width = percent + '%';
    if (progressText) progressText.textContent = 'Прогресс: ' + filled + ' / ' + total;
  }

  function toggleDict() {

  async function searchWord() {
    const input = document.getElementById('dictInput');
    const resultDiv = document.getElementById('dictResult');
    const word = input.value.trim().toLowerCase();

    if (!word) return;

    // 1. Мгновенная проверка в локальном словаре курса
    if (localDict[word]) {
      displayResult(word, localDict[word]);
      return;
    }

    // 2. Если слова нет в курсе, используем быстрый резервный API
    resultDiv.innerHTML = '<div class="dict-loading">Ищу перевод...</div>';
    try {
      const response = await fetch(\`https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=ru&dt=t&q=\${encodeURIComponent(word)}\`);
      const data = await response.json();
      
      if (data && data[0] && data[0][0] && data[0][0][0]) {
        displayResult(word, data[0][0][0]);
      } else {
        resultDiv.innerHTML = \`<div class="dict-error">Перевод не найден. Проверьте написание слова.</div>\`;
      }
    } catch (error) {
      console.error('Dictionary error:', error);
      resultDiv.innerHTML = \`<div class="dict-error">Ошибка соединения. Проверьте интернет.</div>\`;
    }
  }

  function displayResult(word, translation) {
    const resultDiv = document.getElementById('dictResult');
    const safeWord = word.replace(/'/g, "\\\\'"); 
    resultDiv.innerHTML = \`
      <div class="dict-word">
        \${word}
        <button class="dict-audio-btn" onclick="speakWord('\${safeWord}')" title="Прослушать произношение">🔊</button>
      </div>
      <div class="dict-trans">\${translation}</div>
    \`;
  }

  function speakWord(text) {
    if ('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = 'en-US';
      utterance.rate = 0.85;
      utterance.pitch = 1;
      window.speechSynthesis.speak(utterance);
    } else {
      alert('Ваш браузер не поддерживает озвучивание текста.');
    }
  }
