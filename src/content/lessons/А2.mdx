<script>
  // === 1. АВТОСОХРАНЕНИЕ ПРОГРЕССА ===
  const pageKey = 'eng_lesson_' + window.location.pathname;

  function saveField(el){
    const elId = el.id || el.name;
    if (!elId) return;
    let value = el.value;
    if (el.type === 'checkbox' || el.type === 'radio') {
      value = el.checked;
    }
    localStorage.setItem(pageKey + '_' + elId, JSON.stringify(value));
  }

  function restoreFields(){
    document.querySelectorAll('input, textarea, select').forEach(el => {
      const elId = el.id || el.name;
      if (!elId) return;
      const saved = localStorage.getItem(pageKey + '_' + elId);
      if (saved !== null) {
        const value = JSON.parse(saved);
        if (el.type === 'checkbox' || el.type === 'radio') {
          el.checked = value;
        } else {
          el.value = value;
        }
      }
    });
    updateProgress();
  }

  function clearProgress(){
    if (!confirm('Вы уверены, что хотите очистить все ответы в этом уроке?')) return;
    Object.keys(localStorage).forEach(key => {
      if (key.startsWith(pageKey)) {
        localStorage.removeItem(key);
      }
    });
    location.reload();
  }

  // === 2. ПРОГРЕСС-БАР ===
  const progressBar = document.getElementById('progressBar');
  const progressText = document.getElementById('progressText');

  function updateProgress(){
  const fields = document.querySelectorAll('input, textarea, select');
  if (fields.length === 0) {
    progressText.textContent = 'Прогресс: 0 / 0';
    progressBar.style.width = '0%';
    return;
  }
  let filled = 0;
  let total = 0;
  const radioGroups = new Set();
  const radioCorrectGroups = new Set();

  fields.forEach(f => {
    if (f.type === 'radio') {
      // Для radio: считаем группы (по name), а не каждое поле
      radioGroups.add(f.name);
      if (f.checked && f.classList.contains('corr')) {
        radioCorrectGroups.add(f.name);
      }
    } else if (f.type === 'checkbox') {
      total++;
      if (f.checked) filled++;
    } else {
      // input text, textarea, select
      total++;
      if (f.value && f.value.trim() !== '') filled++;
    }
  });

  // Добавляем radio-группы в общий счёт
  total += radioGroups.size;
  filled += radioCorrectGroups.size;

  const percent = total > 0 ? (filled / total) * 100 : 0;
  progressBar.style.width = percent + '%';
  progressText.textContent = 'Прогресс: ' + filled + ' / ' + total;
}

  // === 3. ПОДСВЕТКА АКТИВНОГО РАЗДЕЛА (scroll-based) ===
  let ticking = false;

  function updateActiveSection(){
    // Собираем все элементы, на которые могут указывать кнопки меню
    const allTargets = document.querySelectorAll('[id^="tema"], [id^="t"]');
    let currentId = null;

    // Порог: 35% от высоты экрана сверху
    const threshold = window.innerHeight * 0.35;

    allTargets.forEach(el => {
      const rect = el.getBoundingClientRect();
      // Элемент считается активным, если его верхний край
      // уже пересёк порог сверху
      if (rect.top <= threshold) {
        currentId = el.id;
      }
    });

    if (currentId) {
      highlightActiveButton(currentId);
    }
    ticking = false;
  }

  function highlightActiveButton(sectionId){
    // Снимаем подсветку со всех кнопок
    document.querySelectorAll('.nav-btn-main.current, .nav-btn-sub.current').forEach(btn => {
      btn.classList.remove('current');
    });
    // Находим кнопку с соответствующим data-target и подсвечиваем
    const btn = document.querySelector('[data-target="' + sectionId + '"]');
    if (btn) {
      btn.classList.add('current');

      // Если это кнопка подтемы — автоматически раскрываем её подменю
      if (btn.classList.contains('nav-btn-sub')) {
        const parentSubmenu = btn.closest('.nav-submenu');
        if (parentSubmenu && !parentSubmenu.classList.contains('open')) {
          // Закрываем все остальные подменю
          document.querySelectorAll('.nav-submenu').forEach(el => el.classList.remove('open'));
          document.querySelectorAll('.nav-btn-main').forEach(el => el.classList.remove('active'));
          // Открываем нужное
          parentSubmenu.classList.add('open');
          // Подсвечиваем родительскую кнопку темы
          const parentBtn = parentSubmenu.previousElementSibling;
          if (parentBtn && parentBtn.classList.contains('nav-btn-main')) {
            parentBtn.classList.add('active');
            parentBtn.classList.add('current');
          }
        }
      }
    }
  }

  // Throttled scroll listener для производительности
  window.addEventListener('scroll', () => {
    if (!ticking) {
      requestAnimationFrame(updateActiveSection);
      ticking = true;
    }
  });

  // === 4. НАВИГАЦИЯ ===
  function toggleMenu(){
    const menu = document.getElementById('navMenu');
    const btn = document.querySelector('.nav-toggle');
    const isMobile = window.innerWidth <= 768;

    if(isMobile){
      menu.classList.toggle('visible');
      btn.textContent = menu.classList.contains('visible') ? '✕' : '☰';
    } else {
      menu.classList.toggle('hidden');
      btn.textContent = menu.classList.contains('hidden') ? '☰' : '✕';
    }
  }

  function toggleSubmenu(subId, btnElement){
    document.querySelectorAll('.nav-submenu').forEach(el => {
      if(el.id !== subId) el.classList.remove('open');
    });
    document.querySelectorAll('.nav-btn-main').forEach(el => {
      if(el !== btnElement) el.classList.remove('active');
    });

    const submenu = document.getElementById(subId);
    const isOpen = submenu.classList.contains('open');
    
    if(isOpen){
      submenu.classList.remove('open');
      btnElement.classList.remove('active');
    } else {
      submenu.classList.add('open');
      btnElement.classList.add('active');
    }
  }

  function goTo(id){
    document.getElementById(id).scrollIntoView({behavior:'smooth', block:'center'});
    if(window.innerWidth <= 768){
      const menu = document.getElementById('navMenu');
      menu.classList.remove('visible');
      document.querySelector('.nav-toggle').textContent = '☰';
    }
  }

  // === 5. ЗАПУСК ПРИ ЗАГРУЗКЕ СТРАНИЦЫ ===
  document.addEventListener('input', (e) => {
  if (e.target.matches('input, textarea, select')) {
    saveField(e.target);
    updateProgress();
  }
});

// Добавляем слушатель change для radio и checkbox
document.addEventListener('change', (e) => {
  if (e.target.matches('input[type="radio"], input[type="checkbox"]')) {
    // Для radio: сохраняем состояние всей группы (все варианты с одинаковым name)
    if (e.target.type === 'radio') {
      const name = e.target.name;
      document.querySelectorAll('input[type="radio"][name="' + name + '"]').forEach(r => {
        saveField(r);
      });
    } else {
      saveField(e.target);
    }
    updateProgress();
  }
});

  window.addEventListener('DOMContentLoaded', () => {
    restoreFields();
    // Запускаем подсветку сразу после загрузки
    updateActiveSection();
  });
</script>
