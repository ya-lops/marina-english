      const STORAGE_KEY = 'astro_learning_progress_v1';

      function saveState() {
          try {
              const state = {};
              
              // 1. Текстовые поля, textarea и селекты (по id)
              document.querySelectorAll('input[type="text"], input[type="password"], textarea, select').forEach(el => {

              });
              
              // 3. Кастомные элементы: размытый текст (blur)
              document.querySelectorAll('.blur-wrapper').forEach((el, i) => {

              if (typeof ddPlaced !== 'undefined') state['ddPlaced'] = {...ddPlaced};

              if (typeof matchedPairs !== 'undefined') state['matchedPairs'] = {...matchedPairs};

      function restoreState() {
          try {
              const saved = localStorage.getItem(STORAGE_KEY);
              if (!saved) return;
              const state = JSON.parse(saved);

              // 1. Текстовые поля и селекты
              Object.keys(state).forEach(key => {
                  if (key.startsWith('input_')) {
                      const el = document.getElementById(key.replace('input_', ''));
                      if (el) {
                          el.value = state[key];
                          el.dispatchEvent(new Event('input')); // Обновляет UI (например, скрывает подсказки)

                  }
              });

              // 2. Радиокнопки
              document.querySelectorAll('input[type="radio"]').forEach(el => {
                  const savedVal = state['radio_' + el.name];
                  if (savedVal && el.value === savedVal) {
                      el.checked = true;
                      el.dispatchEvent(new Event('change'));
                  }
              });

                              originalWord.setAttribute('draggable', 'false');
                          }
                      }
                  });

                          if (defEl) defEl.classList.add('matched');

                  if (typeof updateConnections === 'function') {

                          if (defEl) connections.push({wordEl, defEl});
                      });
                      updateConnections();
                  }
              }

              // 6. RU Matching
              if (state.ruMatched && typeof ruMatched !== 'undefined') {
                  Object.assign(ruMatched, state.ruMatched);
                  Object.keys(state.ruMatched).forEach(id => {
                      if (state.ruMatched[id]) {
                          const pair = document.querySelector(\`.ru-pair[data-id="\${id}"]\`);
                          if (pair) {

                              // Скрываем кнопки выбора, показываем результат
                              pair.querySelectorAll('.ru-btn').forEach(btn => btn.classList.add('used'));
                          }
                      }
                  });
              }
              
              // 7. Essay
              if (state.essay) {
                  const essay = document.getElementById('essay');
                  if (essay) {
                      essay.value = state.essay;
                      essay.dispatchEvent(new Event('input'));
                  }
              }

          } catch (e) {

          }
