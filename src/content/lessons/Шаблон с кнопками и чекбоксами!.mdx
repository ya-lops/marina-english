<script>
  // Генерация 30 кнопок с чекбоксами
  const menuContainer = document.getElementById('navMenu');
  for (let i = 1; i <= 30; i++) {
    const btn = document.createElement('button');
    btn.className = 'nav-btn';
    
    btn.onclick = function(e) {
      // Если кликнули по чекбоксу, не делаем прокрутку
      if (e.target.type === 'checkbox') return;
      goTo('email-' + i);
    };

    const checkbox = document.createElement('input');
    checkbox.type = 'checkbox';
    checkbox.className = 'nav-checkbox';
    checkbox.id = 'chk-' + i;
    
    // Останавливаем всплытие, чтобы клик по чекбоксу не триггерил кнопку
    checkbox.onclick = function(e) {
      e.stopPropagation();
    };
    
    // Загрузка состояния из localStorage
    const isCompleted = localStorage.getItem('email_written_' + i) === 'true';
    if (isCompleted) {
      checkbox.checked = true;
      btn.classList.add('completed');
    }

    // Сохранение состояния при изменении
    checkbox.addEventListener('change', function(e) {
      localStorage.setItem('email_written_' + i, e.target.checked);
      if (e.target.checked) {
        btn.classList.add('completed');
      } else {
        btn.classList.remove('completed');
      }
    });

    const label = document.createElement('span');
    label.className = 'nav-label';
    label.textContent = 'Email ' + i;

    btn.appendChild(checkbox);
    btn.appendChild(label);
    menuContainer.appendChild(btn);
  }

  function toggleMenu(){
    const menu = document.getElementById('navMenu');
    const btn = document.querySelector('.nav-toggle');
    const isMobile = window.innerWidth <= 768;

    if(isMobile){
      menu.classList.toggle('visible');
      btn.textContent = menu.classList.contains('visible') ? '✕' : '☰';
    } else {
      menu.classList.toggle('hidden');
      btn.textContent = menu.classList.contains('hidden') ? '☰' : '✕';
    }
  }

  function goTo(id){
    document.getElementById(id).scrollIntoView({behavior:'smooth', block:'center'});
    if(window.innerWidth <= 768){
      const menu = document.getElementById('navMenu');
      menu.classList.remove('visible');
      document.querySelector('.nav-toggle').textContent = '☰';
    }
  }
</script>
