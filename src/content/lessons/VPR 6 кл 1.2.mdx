<script>
  // === 1. АВТОСОХРАНЕНИЕ ПРОГРЕССА ===
  const pageKey = 'eng_lesson_' + window.location.pathname;

  function saveField(el){
    const elId = el.id || el.name;
    if (!elId) return;
    let value = el.value;
    if (el.type === 'checkbox' || el.type === 'radio') {
      value = el.checked;
    }
    localStorage.setItem(pageKey + '_' + elId, JSON.stringify(value));
  }

  function restoreFields(){
    document.querySelectorAll('input, textarea, select').forEach(el => {
      const elId = el.id || el.name;
      if (!elId) return;
      const saved = localStorage.getItem(pageKey + '_' + elId);
      if (saved !== null) {
        const value = JSON.parse(saved);
        if (el.type === 'checkbox' || el.type === 'radio') {
          el.checked = value;
        } else {
          el.value = value;
        }
      }
    });
    updateProgress();
  }

  function updateProgress() {
  let filled = 0;
  let total = 0;

  // 1. Считаем радиокнопки (по группам name)
  const radios = document.querySelectorAll('input[type="radio"]');
  const radioGroups = new Set();
  radios.forEach(r => {
    radioGroups.add(r.name);
    if (r.checked && r.classList.contains('corr')) {
      filled++;
    }
  });
  total += radioGroups.size;

  // 2. Считаем специальные трекеры прогресса (текстовые поля и скрытые инпуты)
  const trackers = document.querySelectorAll('.progress-tracker');
  total += trackers.length;
  trackers.forEach(tracker => {
    if (tracker.dataset.status === 'correct') {
      filled++;
    }
  });

  const percent = total > 0 ? (filled / total) * 100 : 0;
  const progressBar = document.getElementById('progressBar');
  const progressText = document.getElementById('progressText');
  if (progressBar) progressBar.style.width = percent + '%';
  if (progressText) progressText.textContent = 'Прогресс: ' + filled + ' / ' + total;
}

// Эта функция просто сбрасывает статус поля, но НЕ вызывает updateProgress!
// Поэтому прогресс-бар не дёргается при вводе.
function resetTrackerStatus(el) {
  if (el.classList.contains('progress-tracker')) {
    el.dataset.status = 'none';
    el.style.borderColor = '#cbd5e1';
    el.style.background = '#fff';
  }
}

// === ГЛОБАЛЬНЫЙ СБРОС ВСЕХ ОТВЕТОВ ===
function clearProgress() {
  if (!confirm('Вы уверены, что хотите очистить все ответы в этом уроке? Это действие нельзя отменить.')) return;
  const pageKey = 'eng_lesson_' + window.location.pathname;
  Object.keys(localStorage).forEach(key => {
    if (key.startsWith(pageKey)) {
      localStorage.removeItem(key);
    }
  });
  location.reload();
}


  // === 3. ПОДСВЕТКА АКТИВНОГО РАЗДЕЛА (Исправленная логика) ===
  let ticking = false;

  function updateActiveSection(){
    const allTargets = document.querySelectorAll('[id^="tema"], [id^="t"]');
    const threshold = window.innerHeight * 0.35;
    
    let activeEl = null;
    
    // 1. Ищем элемент, который физически находится на уровне порога (35% сверху)
    allTargets.forEach(el => {
      const rect = el.getBoundingClientRect();
      if (rect.top <= threshold && rect.bottom >= threshold) {
        activeEl = el;
      }
    });
    
    // 2. Если не нашли (например, короткие блоки и мы прокручиваем между ними), 
    // берем последний, чей верх уже пересек порог
    if (!activeEl) {
      allTargets.forEach(el => {
        const rect = el.getBoundingClientRect();
        if (rect.top <= threshold) {
          activeEl = el;
        }
      });
    }
    
    if (activeEl) {
      highlightActiveButton(activeEl.id);
    }
    ticking = false;
  }

  function highlightActiveButton(sectionId){
    document.querySelectorAll('.nav-btn-main.current').forEach(btn => {
      btn.classList.remove('current');
    });
    const btn = document.querySelector('[data-target="' + sectionId + '"]');
    if (btn && btn.classList.contains('nav-btn-main')) {
      btn.classList.add('current');
    }
  }

  window.addEventListener('scroll', () => {
    if (!ticking) {
      requestAnimationFrame(updateActiveSection);
      ticking = true;
    }
  });

  // === 4. НАВИГАЦИЯ ===
  function toggleMenu(){
    const menu = document.getElementById('navMenu');
    const btn = document.querySelector('.nav-toggle');
    const isMobile = window.innerWidth <= 768;

    if(isMobile){
      menu.classList.toggle('visible');
      btn.textContent = menu.classList.contains('visible') ? '✕' : '☰';
    } else {
      menu.classList.toggle('hidden');
      btn.textContent = menu.classList.contains('hidden') ? '☰' : '✕';
    }
  }

  function goTo(id){
    document.getElementById(id).scrollIntoView({behavior:'smooth', block:'center'});
    if(window.innerWidth <= 768){
      const menu = document.getElementById('navMenu');
      menu.classList.remove('visible');
      document.querySelector('.nav-toggle').textContent = '☰';
    }
  }

  function scrollToTop(){
    window.scrollTo({ top: 0, behavior: 'smooth' });
    if(window.innerWidth <= 768){
      const menu = document.getElementById('navMenu');
      menu.classList.remove('visible');
      document.querySelector('.nav-toggle').textContent = '☰';
    }
  }

  // === 5. ЗАПУСК ПРИ ЗАГРУЗКЕ СТРАНИЦЫ ===
  document.addEventListener('input', (e) => {
    if (e.target.matches('input, textarea, select')) {
      saveField(e.target);
    }
  });

  document.addEventListener('change', (e) => {
    if (e.target.matches('input[type="radio"], input[type="checkbox"]')) {
      if (e.target.type === 'radio') {
        const name = e.target.name;
        document.querySelectorAll('input[type="radio"][name="' + name + '"]').forEach(r => {
          saveField(r);
        });
      } else {
        saveField(e.target);
      }
      updateProgress();
    }
  });

  window.addEventListener('DOMContentLoaded', () => {
    restoreFields();
    updateActiveSection();
  });
</script>
